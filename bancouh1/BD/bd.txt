-- creacion de de la baes de datos BANCOUH
create database BancoUH

use BancoUH

-- creacion de las tabla

create table Usuarios
(
   correo varchar(50) primary key,
   clave varchar(50) not null,
   nombre varchar(100)
)


-- Insertar usuarios
Insert into USuarios (correo, clave, nombre) values (@correo,@clave,@nombre'),
                                                    ('Jose@uh.com','456','Jose Ramirez')
Insert into USuarios (correo, clave, nombre) values ('Suana@uh.com','487','Susana Jimenez')

-- consultar usuarios

Select * from usuarios
Select correo, clave, nombre from usuarios
select correo from usuarios
